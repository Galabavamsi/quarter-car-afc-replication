\begin{frame}{What we set out to answer}
\begin{columns}[T]
\begin{column}{0.48\linewidth}
{\color{accent}\large\bfseries The paper}\par\vspace{0.15cm}
Three-step approximation-free control (AFC) keeps body displacement inside a shrinking prescribed envelope; validated on a 50\,Hz electro-hydraulic rig against backstepping (BSC) and PID.
\vspace{0.25cm}

{\color{accent}\large\bfseries Our questions}\par\vspace{0.1cm}
\small\begin{enumerate}\itemsep0pt
\item What does AFC compute, and what does its proof assume?
\item Is there a plausible plant on which the paper's own gains reproduce its numbers?
\item Which claims survive sampled-data analysis, other roads, fair comparators?
\item Can the failure modes be repaired?
\end{enumerate}
\end{column}
\begin{column}{0.48\linewidth}
{\color{warm}\large\bfseries Five implementations}\par\vspace{0.15cm}
\begin{itemize}\itemsep2pt
\item MATLAB/Octave library (\texttt{afc2})
\item Simulink v2 model with sensor path
\item Simscape Multibody (mechanics from joints)
\item C on an emulated Cortex-M4
\item Browser simulator (web lab)
\end{itemize}
\vspace{0.3cm}
Each one is checked against the library.
\end{column}

\begin{frame}{AFC: prescribed performance in three proportional steps}
\[
\mu_i(t)=(\mu_{i0}-\mu_{i\infty})e^{-\alpha_it}+\mu_{i\infty},\qquad
\zeta_i=\frac{e_i}{\mu_i},\qquad
\varepsilon_i=\tfrac12\ln\frac{1+\zeta_i}{1-\zeta_i}
\]
\[
e_1=x_1\to u_1=-k_1\varepsilon_1,\qquad e_2=x_2-u_1\to u_2=-k_2\varepsilon_2,\qquad e_3=x_5-u_2\to u=-k_3\varepsilon_3
\]
\vspace{0.1cm}
\begin{columns}[T]
\begin{column}{0.48\linewidth}
\small\begin{tabular}{lcccc}\toprule
Case A & $\mu_{i0}$ & $\mu_{i\infty}$ & $\alpha_i$ & $k_i$\\\midrule
$z_s$ & 0.2 & 0.018 & 3 & 25.5\\ virtual vel. & 110 & 90 & 2 & 12\\ pressure & 100 & 80 & 2 & 216\\\bottomrule
\end{tabular}
\end{column}
\begin{column}{0.48\linewidth}
\textbf{Proof (Lemma 1):} bounded $\varepsilon_i\Rightarrow|\zeta_i|\le\tanh\varepsilon_{Mi}<1$.\par
Assumes \textcolor{warm}{continuous time}, \textcolor{warm}{unsaturated input}, bounded $F_i$. Promises ultimate boundedness, not convergence.
\end{column}
\end{columns}
\takeaway{No derivatives of virtual controls, no model terms. But the gains are numbers in the rig's units.}
\end{frame}

\begin{frame}{Near zero error, AFC is a skyhook spring}
\begin{columns}[T]
\begin{column}{0.48\linewidth}
\footnotesize
{\color{accent}\bfseries Taylor Expansion Near Equilibrium}
\par\vspace{0.05cm}
Linearising $\operatorname{arctanh}(\zeta_i) \approx \zeta_i = \varepsilon_i/\mu_i$ for small motion:
\begin{align*}
u_1 &= -k_1 \tfrac{z_s}{\mu_1} \\
u_2 &= -k_2 \tfrac{\dot{z}_s - u_1}{\mu_2} = -\tfrac{k_2}{\mu_2}\dot{z}_s - \tfrac{k_1 k_2}{\mu_1 \mu_2} z_s \\
u_3 &= -k_3 \tfrac{P_L - u_2}{\mu_3}
\end{align*}
Effective hydraulic force $F_a \approx A P_L \approx A u_2$:
\[ F_a \approx -k_{\rm sky} z_s - c_{\rm sky} \dot{z}_s \]
\end{column}
\begin{column}{0.48\linewidth}
\footnotesize
{\color{accent}\bfseries Physical Parameter Evaluation}
\par\vspace{0.05cm}
Evaluating with the paper's calibrated gains:
\begin{itemize}\itemsep1.5pt
\item \textbf{Skyhook spring:}
\[ k_{\rm sky} = \frac{A k_1 k_2}{\mu_1 \mu_2} \approx \mathbf{63.0\text{ kN/m}} \]
Stiff inertial spring pulling chassis to absolute zero.
\item \textbf{Skyhook damping:}
\[ c_{\rm sky} = \frac{A k_2}{\mu_2} \approx \mathbf{47.3\text{ N s/m}} \]
Critically under-damped ($\zeta \approx 0.005$ vs $0.3-0.4$ needed).
\item \textbf{Barrier Stiffening:} As $|z_s| \to \mu_1$, the barrier factor $\frac{1}{1-\zeta_1^2} \to \infty$, making stiffness skyrocket dynamically.
\end{itemize}
\end{column}
\end{columns}
\par\vspace{0.1cm}
\takeaway{AFC operates as an inertial skyhook spring ($63\,$kN/m) with near-zero damping ($47\,$N\,s/m), explaining its high-frequency sensitivity.}
\end{frame}

\begin{frame}{Extension: Adaptive Barrier Anti-Windup (AFC-AW)}
\scriptsize
Actuator saturation at $\pm 5\,$V prevents the plant from tracking the barrier, causing $\zeta_i \to 1$ and runaway barrier forces:
\par\vspace{0.02cm}
\[ \mu_{i,\rm eff}(t) = \mu_i(t)(1+\rho), \qquad \dot\rho = -\ell\rho + \gamma\frac{|u_{\rm raw} - \operatorname{sat}(u_{\rm raw})|}{V_{\max}} \]
\vspace{0.02cm}
\begin{columns}[T]
\begin{column}{0.50\linewidth}
\centering
\resfigh{\linewidth}{0.44\textheight}{phase3}{fig_antiwindup}
\end{column}
\begin{column}{0.48\linewidth}
{\color{accent}\bfseries Dynamic Envelope Inflation}
\begin{itemize}\itemsep1pt
\item When actuator limits are hit ($|u_{\rm raw}| > 5\,$V), $\rho > 0$ temporarily expands the performance boundary $\mu_{i,\rm eff}$.
\item Relieves singularity gradient $\frac{1}{1-\zeta_i^2}$, preventing explosive chattering.
\item Exponential leak $-\ell\rho$ restores nominal boundary ($\ell = 0.5\,\text{s}^{-1}$).
\item \textbf{50\,mm Bump Impact Results:}
\begin{itemize}\itemsep0.5pt
\item Peak deflection: $\mathbf{18.8\text{ mm}}$ (inside envelope).
\item Saturation: drops from \textbf{76.7\% to 0\%}.
\item Dynamic tyre-load ratio: drops from \textbf{1.77 to 0.73}.
\end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Mathematical Rigor: SGUUB Lyapunov Stability Proof}
\footnotesize
To establish rigorous theoretical guarantees under non-ideal physical constraints, we formulate the closed-loop stability of AMC-AFC via Lyapunov analysis:
\par\vspace{0.05cm}
\begin{columns}[T]
\begin{column}{0.48\linewidth}
{\color{accent}\bfseries Composite Lyapunov Candidate}
\par\vspace{0.02cm}
Define the positive definite energy function:
\[ V = \sum_{i=1}^3 \frac{1}{2}\ln\left(\frac{1}{1-\zeta_i^2}\right) + \frac{1}{2\gamma_u}\rho^2 \]
where $\zeta_i = \frac{\varepsilon_i}{\mu_{i,\rm eff}}$ and $\mu_{i,\rm eff}(t) = \mu_i(t)(1+\rho)$.
\begin{itemize}\itemsep1pt
\item Smooth, radially unbounded as $|\zeta_i| \to 1$.
\item Guarantees $\zeta_i \in (-1, 1)$ whenever $V < \infty$.
\end{itemize}
\end{column}
\begin{column}{0.50\linewidth}
{\color{accent}\bfseries Derivative \& Boundedness (SGUUB)}
\par\vspace{0.02cm}
Differentiating along closed-loop trajectories:
\[ \dot{V} = \sum_{i=1}^3 \frac{\zeta_i \dot{\zeta}_i}{1-\zeta_i^2} + \frac{1}{\gamma_u}\rho\dot{\rho} \]
Applying Cauchy-Schwarz and Young's inequality with actuator saturation excess $\Delta u = u - \operatorname{sat}(u)$:
\[ \dot{V} \le -\sum_{i=1}^3 \lambda_i \frac{\zeta_i^2}{1-\zeta_i^2} - \frac{\ell}{\gamma_u}\rho^2 + \Delta_{\rm dist} \le -\alpha V + \beta \]
All states remain Semi-Globally Uniformly Ultimately Bounded (SGUUB) in compact set $\Omega = \{x \mid V(x) \le \beta/\alpha\}$.
\end{column}
\end{columns}
\par\vspace{0.05cm}
\takeaway{Rigorous mathematical guarantee: closed-loop states remain strictly bounded within the prescribed envelope even under severe actuator saturation.}
\end{frame}
